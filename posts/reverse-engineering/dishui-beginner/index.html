    //PE头
    pFileHeader = (PIMAGE_FILE_HEADER)((DWORD)pNTHeader + 4);
    //节区数
    int sectionNums = pFileHeader->NumberOfSections;
    //OptionHeader大小
    int sizeOfOptionalHeader = pFileHeader->SizeOfOptionalHeader;
    //可选PE头
    pOptionalHeader = (PIMAGE_OPTIONAL_HEADER)((DWORD)pFileHeader + IMAGE_SIZEOF_FILE_HEADER);
    //第一个节区头
    pSectionHeader = (PIMAGE_SECTION_HEADER)((DWORD)pOptionalHeader + sizeOfOptionalHeader);

    //计算相对偏移 dwRva 就是相对偏移了 va - pImageBuffer
    //遍历节区表进行比较
    for(int i=0;i<sectionNums;i++){
        //此时name是正常的 没有占八位 可以直接输出
        // NAME

        if(dwRva >= (DWORD)(pSectionHeader->VirtualAddress) && dwRva < (DWORD)(pSectionHeader->VirtualAddress) + (DWORD)(pSectionHeader->Misc.VirtualSize) )
            break;
        pSectionHeader = (PIMAGE_SECTION_HEADER)((DWORD)pSectionHeader + IMAGE_SIZEOF_SECTION_HEADER);
    }
    //计算 RAW
    DWORD RAW = dwRva - pSectionHeader->VirtualAddress + pSectionHeader->PointerToRawData;
    return RAW;
}

  //节区数
  int sectionNums = pFileHeader->NumberOfSections;

  //判断rva是否在 headers+节表处
  if (dwRva < pOptionalHeader32->SizeOfHeaders) {
    //不需要进行处理 没有拉伸 在文件和内存中都是一样的
    return dwRva;
  }
  //计算相对偏移 dwRva 就是相对偏移了 va - pImageBuffer
  //遍历节区表进行比较
  for (int i = 0; i < sectionNums; i++) {
    //此时name是正常的 没有占八位 可以直接输出
    // NAME
    // printf("%s\n",&(pSectionHeader->Name));
    if (dwRva >= (DWORD)(pSectionHeader->VirtualAddress) &&
        dwRva < (DWORD)(pSectionHeader->VirtualAddress) +
                    (DWORD)(pSectionHeader->Misc.VirtualSize))
      break;
    pSectionHeader = (PIMAGE_SECTION_HEADER)((DWORD)pSectionHeader +
                                             IMAGE_SIZEOF_SECTION_HEADER);
  }
  //计算 RAW
  DWORD RAW =
      dwRva - pSectionHeader->VirtualAddress + pSectionHeader->PointerToRawData;
  pSectionHeader = originalSectionHeader; 
  return RAW;
}

083 Win32 加密壳_项目说明(√)

壳子分为两类 (加密壳 压缩壳? 不是)

加密壳可以分为两类

  • 二进制壳

    并不改变里面指令(硬编码)的含义 只是把PE文件做一个扭曲(变形)

    变形之后把一些关键的部位进行加密

    加密之后就难以分析

    但是所有的二进制壳都有一个问题 执行的时候 在内存中 一定是它展开的模样

    脱壳的过程就是跟着指令到一个关键部位 内存中的数据已经展开了 把内存中的数据给它dump出去，dump完之后，修改入口点(修改入口点 dump出去)

    (大部分壳都是拿win32汇编写的 现在还不会)

  • 源代码加密 指令壳

    指令壳 有把一个汇编指令变多个

  1. 加壳程序的编写

    加密后添加到壳子后面

    为什么此时不用移动表?

    这种思路: 要加密PE文件 新增一个节 把需要保存的各种表移动到这个节中(因为一个进程运行起来的过程中 需要修复IAT表 要是把表加密了就不行了) 然后进行加密

    解壳

    新增的代码必须是一段完整的shellcode 机器码 需要什么dll自己去加载(必须保证这段代码贴在哪里都能跑 需要大量事件) 所以需要用win32汇编来写

    我们的思路(可以不用移动表)

    对整个文件进行加密 想多复杂就多复杂

    加密后 不过是一段二进制代码 进行了免杀(文件免杀) (这种壳子秒破 取决于壳子复不复杂)

    特征码(提取一块硬编码)

    双击之后跑起来的程序是shell.exe 最复杂的程序全在解壳的过程中

  2. 解壳过程的编写

    1. 读取主模块的数据(加密后二进制数据)

    2. 解密 得到原来的PE文件

    3. 以挂起的形式创建进程,CreateProcess 要创建进程在哪里?

      要创建的进程就是壳子本身 (以挂起形式创建 只有一个外壳) 换句话说 只要它的一个空间

    4. 获取外壳程序的Context,后面要用

    5. 卸载外壳程序(以挂起方式创建 得到空间后 把外壳程序卸载)

    6. 在指定的位置分配空间:位置就是src的ImageBase 大小就是src的SizeOfImage

    7. 如果成功，将Src的PE文件拉伸 复制到该空间中

    8. 如果申请空间失败，判断该Src程序是否有重定位表(申请失败 说明指定的imageBase被占用了 所以需要使用重定位表来进行移位 没有重定位表 里面的全局变量都找不到了)

      如果申请空间失败，但又重定位表，在任意位置申请空间，然后将PE文件复制、拉伸、修复重定位表

    9. 如果第六部申请空间失败，并且没有重定位表，直接返回，失败

    10. 修改外壳程序的Context:

      将Context的ImageBase 改为 Src的ImagBase(可能是申请空间失败后修改的值)

      将Context的OEP 改为 Src的OEP

    11. 设置Context 并恢复主线程

    12. 终止外壳程序，解壳程序结束
